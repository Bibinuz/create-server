# Xmx and Xms set the maximum and minimum RAM usage, respectively.
# By setting -Xms lower than -Xmx, Java can dynamically allocate and release memory.

# Initial memory allocation (starts small when idle)
-Xms1G

# Maximum memory allocation (can grow up to 28GB under load)
-Xmx28G

# Use Generational ZGC for low-latency garbage collection
-XX:+UseZGC
-XX:+ZGenerational

# Automatically return unused memory back to the OS after 60 seconds of idling
-XX:ZUncommitDelay=60
